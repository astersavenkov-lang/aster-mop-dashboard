def verify_telegram(init_data: str) -> bool:
    try:
        vals = dict(x.split('=', 1) for x in init_data.split('&') if '=' in x)
        check_hash = vals.pop('hash', '')
        data_check = '\n'.join(f'{k}={v}' for k, v in sorted(vals.items()))
        secret = hmac.new(b'WebAppData', BOT_TOKEN.encode(), hashlib.sha256).digest()
        result = hmac.new(secret, data_check.encode(), hashlib.sha256).hexdigest()
        return hmac.compare_digest(result, check_hash)
    except Exception:
        return True  # В режиме разработки пропускаем

# ── Парсинг Excel ────────────────────────────
def parse_xlsx(file_stream):
    wb = load_workbook(file_stream, read_only=True, data_only=True)
    ws = wb.active
    rows = list(ws.iter_rows(values_only=True))
    if not rows:
        return None
    hdr = rows[0]
    def ci(name):
        return next((i for i, h in enumerate(hdr) if h and name in str(h)), -1)
    idx = {
        'branch': ci('Филиал'), 'mop': ci('МОП'),
        'rl_calls': ci('Звонки'), 'rl_inet': ci('Интернет'),
        'rl_walk': ci('Спонт'), 'rl_all': ci('РЛ (Все)'),
        'vis_all': ci('Визиты (Все)'), 'td': ci('Тест'),
        'sales_all': ci('Продажи (Все)'), 'lost_all': ci('Упущены (Все)'),
    }
    EXCLUDED = ['Hyundai Бухара','Chery Ташкент Алмазар',
                'Haval Ташкент Алмазар','Lada Бухара','Lada Наманган']
    branch_map = {}
    for row in rows[1:]:
        br_name = row[idx['branch']] if idx['branch'] >= 0 else None
        if not br_name: continue
        if any(ex in str(br_name) for ex in EXCLUDED): continue
        mop_name = (row[idx['mop']] if idx['mop'] >= 0 else None) or 'Не назначен'
        def n(i): return float(row[i] or 0) if i >= 0 and i < len(row) else 0
        rl, vis, sales, lost = n(idx['rl_all']), n(idx['vis_all']), n(idx['sales_all']), n(idx['lost_all'])
        calls, inet, walk = n(idx['rl_calls']), n(idx['rl_inet']), n(idx['rl_walk'])
        conv = f"{sales/rl*100:.1f}%" if rl > 0 else "0%"
        vis_pct = f"{vis/rl*100:.1f}%" if rl > 0 else "0%"
        lost_pct = f"{lost/rl*100:.1f}%" if rl > 0 else "0%"
        if br_name not in branch_map:
            branch_map[br_name] = {'name': str(br_name), 'mops': []}
        branch_map[br_name]['mops'].append({
            'name': str(mop_name), 'rl': int(rl), 'visits': int(vis),
            'sales': int(sales), 'lost': int(lost), 'calls': int(calls),
            'inet': int(inet), 'walk': int(walk),
            'conv': conv, 'vis_pct': vis_pct, 'lost_pct': lost_pct
        })
    branches = []
    for br in branch_map.values():
        rl = sum(m['rl'] for m in br['mops'])
        vis = sum(m['visits'] for m in br['mops'])
        sales = sum(m['sales'] for m in br['mops'])
        lost = sum(m['lost'] for m in br['mops'])
        conv_v = sales/rl*100 if rl > 0 else 0
        status = 'zero' if sales == 0 else 'top' if conv_v >= 5 else 'mid' if conv_v >= 2 else 'low'
        branches.append({**br,
            'rl': rl, 'visits': vis, 'sales': sales, 'lost': lost,
            'conv': f"{conv_v:.1f}%",
            'lost_pct': f"{lost/rl*100:.1f}%" if rl > 0 else "0%",
            'mop_cnt': len(br['mops']), 'status': status
        })
    return sorted(branches, key=lambda x: -x['sales'])

# ── API endpoints ─────────────────────────────
@app.route('/')
def index():
    return send_from_directory('static', 'index.html')

@app.route('/api/history', methods=['GET'])
def get_history():
    conn = get_db()
    rows = conn.execute('SELECT date, label, data, saved_at FROM snapshots ORDER BY date DESC').fetchall()
    conn.close()
    return jsonify([{'date': r[0], 'label': r[1], 'branches': json.loads(r[2]), 'savedAt': r[3]} for r in rows])

@app.route('/api/upload', methods=['POST'])
def upload():
    if 'file' not in request.files:
        return jsonify({'error': 'Нет файла'}), 400
    file = request.files['file']
    date_str = request.form.get('date', datetime.now().strftime('%Y-%m-%d'))
    label = request.form.get('label', datetime.now().strftime('%d %B %Y'))
    try:
        branches = parse_xlsx(file.stream)
        if not branches:
            return jsonify({'error': 'Не удалось распознать файл'}), 400
        conn = get_db()
        conn.execute(
            'INSERT OR REPLACE INTO snapshots (date, label, data, saved_at) VALUES (?,?,?,?)',
            (date_str, label, json.dumps(branches, ensure_ascii=False), datetime.now().isoformat())
        )
        conn.commit(); conn.close()
        return jsonify({'ok': True, 'branches': len(branches), 'label': label})
    except Exception as e:
        return jsonify({'error': str(e)}), 500

@app.route('/api/snapshot/', methods=['DELETE'])
def delete_snapshot(date):
    conn = get_db()
    conn.execute('DELETE FROM snapshots WHERE date=?', (date,))
    conn.commit(); conn.close()
    return jsonify({'ok': True})
